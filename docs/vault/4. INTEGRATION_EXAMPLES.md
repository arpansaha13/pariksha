# Vault Integration Examples for Pariksha Services

This document provides practical examples for integrating Vault into your Pariksha microservices.

## Go Client Implementation

### Basic Vault Client Setup

```go
package vault

import (
    "context"
    "fmt"
    "log"

    "github.com/hashicorp/vault/api"
    "github.com/hashicorp/vault/api/auth/kubernetes"
)

// Client wraps the Vault API client
type Client struct {
    client *api.Client
}

// NewClient creates and authenticates a new Vault client
func NewClient(ctx context.Context, vaultAddr, role string) (*Client, error) {
    // Create the base client
    client, err := api.NewClient(&api.Config{
        Address: vaultAddr,
    })
    if err != nil {
        return nil, fmt.Errorf("unable to initialize Vault client: %w", err)
    }

    // Authenticate using Kubernetes auth
    k8sAuth, err := kubernetes.NewKubernetesAuth(role)
    if err != nil {
        return nil, fmt.Errorf("unable to initialize Kubernetes auth method: %w", err)
    }

    authInfo, err := client.Auth.Login(ctx, k8sAuth)
    if err != nil {
        return nil, fmt.Errorf("unable to login with Kubernetes auth: %w", err)
    }

    // Verify we got a token
    if authInfo.Auth.ClientToken == "" {
        return nil, fmt.Errorf("failed to receive client token")
    }

    log.Printf("Successfully authenticated to Vault with role: %s", role)

    return &Client{client: client}, nil
}

// GetSecret retrieves a secret from Vault's KV v2 engine
func (c *Client) GetSecret(ctx context.Context, path string) (map[string]interface{}, error) {
    secret, err := c.client.KVv2("secret").Get(ctx, path)
    if err != nil {
        return nil, fmt.Errorf("unable to read secret from %s: %w", path, err)
    }

    if secret == nil {
        return nil, fmt.Errorf("secret not found at %s", path)
    }

    return secret.Data["data"].(map[string]interface{}), nil
}

// GetSecretString retrieves a single string value from Vault
func (c *Client) GetSecretString(ctx context.Context, path, key string) (string, error) {
    data, err := c.GetSecret(ctx, path)
    if err != nil {
        return "", err
    }

    value, ok := data[key]
    if !ok {
        return "", fmt.Errorf("key %s not found in secret", key)
    }

    strValue, ok := value.(string)
    if !ok {
        return "", fmt.Errorf("value is not a string")
    }

    return strValue, nil
}

// Close closes the Vault client connection
func (c *Client) Close() error {
    c.client.ClearToken()
    return nil
}
```

### Usage in Your Service

#### Auth Service Example

```go
package main

import (
    "context"
    "log"
    "os"

    "yourmodule/vault"
)

func main() {
    vaultAddr := os.Getenv("VAULT_ADDR")
    if vaultAddr == "" {
        vaultAddr = "http://vault-service:8200"
    }

    ctx := context.Background()

    // Initialize Vault client
    vaultClient, err := vault.NewClient(ctx, vaultAddr, "auth")
    if err != nil {
        log.Fatalf("Failed to initialize Vault client: %v", err)
    }
    defer vaultClient.Close()

    // Retrieve database credentials
    dbPassword, err := vaultClient.GetSecretString(ctx, "pariksha/auth", "db_password")
    if err != nil {
        log.Fatalf("Failed to retrieve database password: %v", err)
    }

    dbUser, err := vaultClient.GetSecretString(ctx, "pariksha/auth", "db_user")
    if err != nil {
        log.Fatalf("Failed to retrieve database user: %v", err)
    }

    // Use credentials to connect to database
    dsn := fmt.Sprintf(
        "postgres://%s:%s@auth-db:5432/auth_db?sslmode=disable",
        dbUser,
        dbPassword,
    )

    // Connect to database with dsn...
}
```

```yaml

kind: Deployment
metadata:
  name: auth-deployment

    app: auth
    component: api

        - name: auth
          image: auth_service:latest

            - name: VAULT_ADDR
```

## Token Management

### Automatic Token Renewal

```go
package vault

import (
    "context"
    "time"
)

// StartTokenRenewal starts a goroutine to renew the token periodically
func (c *Client) StartTokenRenewal(ctx context.Context, interval time.Duration) {
    go func() {
        ticker := time.NewTicker(interval)
        defer ticker.Stop()

        for {
            select {
            case <-ticker.C:
                renewer, err := c.client.Auth.NewLifetimeRenewal()
                if err != nil {
                    log.Printf("Error creating token renewer: %v", err)
                    continue
                }

                _, err = renewer.Renew()
                if err != nil {
                    log.Printf("Error renewing token: %v", err)
                }
            case <-ctx.Done():
                return
            }
        }
    }()
}
```

## Error Handling

```go
package vault

import (
    "errors"
    "fmt"

    "github.com/hashicorp/vault/api"
)

// HandleVaultError provides appropriate error messages for Vault operations
func HandleVaultError(err error) error {
    if err == nil {
        return nil
    }

    // Check for specific Vault errors
    var respErr *api.ResponseError
    if errors.As(err, &respErr) {
        switch respErr.StatusCode {
        case 401:
            return fmt.Errorf("unauthorized to access Vault: %w", err)
        case 403:
            return fmt.Errorf("forbidden: insufficient permissions in Vault policy: %w", err)
        case 404:
            return fmt.Errorf("secret not found in Vault: %w", err)
        }
    }

    return err
}
```

## Testing

### Mock Vault Client for Tests

```go
package vault

import (
    "context"
)

// MockClient implements a mock Vault client for testing
type MockClient struct {
    Secrets map[string]map[string]interface{}
}

func (m *MockClient) GetSecret(ctx context.Context, path string) (map[string]interface{}, error) {
    if secret, ok := m.Secrets[path]; ok {
        return secret, nil
    }
    return nil, fmt.Errorf("secret not found at %s", path)
}

func (m *MockClient) GetSecretString(ctx context.Context, path, key string) (string, error) {
    secret, err := m.GetSecret(ctx, path)
    if err != nil {
        return "", err
    }
    value, ok := secret[key].(string)
    if !ok {
        return "", fmt.Errorf("key not found or not a string")
    }
    return value, nil
}

func (m *MockClient) Close() error {
    return nil
}
```
