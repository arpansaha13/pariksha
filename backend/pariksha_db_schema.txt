Table users {
  id bigint [primary key]
  username varchar(255) [not null, unique]
  email varchar(255) [not null, unique]
  password varchar(255) [null]
  first_name varchar(255) [null]
  last_name varchar(255) [null]
  verified boolean [not null, default: false]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table exams {
  id bigint [primary key]
  title varchar
  starts_at timestamp [not null]
  ends_at timestamp [not null]
  created_by bigint [not null]
  type varchar(16) [default: 'LINK', note: 'Values can be "LINK" or "INVITE"']
  max_candidates_count integer [not null, note: 'Maximum number of candidates that can take the exam']
  max_score integer [default: 0]
  duration_minutes smallint [not null, note: 'max duration_minutes can be 1440']
  paper_hash varchar(64)
  participant_counts jsonb [default: '{"unattended":0,"invited":0,"started":0,"ended":0}']
  hash varchar(64) [not null, unique]
  deleted_at timestamp
}

Table papers {
  id bigint [primary key]
  title varchar(255) [not null, default: 'Untitled Paper']
  max_score integer [default: 0, note: 'Maximum score that can be awarded in this paper - Sum of individual scores of each question']
  duration_minutes smallint [not null, note: 'max duration_minutes can be 1440']
  question_counts jsonb [default: '{"mcq":0,"subjective":0,"coding":0}', note: 'Count of each question type']
  created_by bigint [not null]
  hash varchar(64) [not null, unique]
  deleted_at timestamp
}

Table paper_permissions {
  paper_id bigint [primary key]
  user_id bigint [primary key]
  permissions smallint [not null, note: 'Bit flags for READ and WRITE permissions']
  deleted_at timestamp
}

Table questions {
  id bigint [primary key]
  category_id bigint [not null]
  question jsonb [not null]
  order smallint [not null]
  type smallint [not null, note: 'MCQ=1, Subjective=2, Coding=3']
  tags jsonb [default: null]
  paper_id bigint
  max_score smallint [not null, note: 'max_score ranges from 0 to 1000']
  correct_answer text
  locked boolean [not null, default: false, note: 'Indicates if the question is used in an exam and is locked for editing']
  hash varchar(64) [not null, unique]
  deleted_at timestamp
}

Table exam_participants {
  id bigint [primary key]
  exam_id bigint [not null]
  user_id bigint [not null]
  score_awarded integer [not null, note: 'Total score obtained in this exam by this candidate - Sum of individual scores of each answer']
  status smallint [default: 1, not null]
  started_at timestamp
  ended_at timestamp
  scheduled_end_time timestamp
}

Table answers {
  id bigint [primary key]
  exam_participant_id bigint [not null]
  question_id bigint [not null]
  answer jsonb
  score_awarded smallint [not null, default: 0]
  evaluated boolean [not null, default: false]
}

Table exam_questions {
  id bigint [primary key]
  exam_id bigint [not null]
  type smallint [not null, note: 'MCQ=1, Subjective=2, Coding=3']
  question_id bigint [not null]
  category_id bigint [not null]
  order smallint [not null]
  max_score smallint [not null]

  // Composite unique key
  indexes {
    (exam_id, question_id) [unique]
  }
}

Table exam_categories {
  id bigint [primary key]
  exam_id bigint [not null]
  category_id bigint [not null]
  order smallint [not null, default: 0]
}

Table sessions {
  key uuid [primary key]
  token varchar(255) [not null]
  expires_at timestamp [not null]
  csrf_token varchar(32) [not null]
}

Table question_categories {
  id bigint [primary key]
  paper_id bigint
  name varchar(255) [not null]
  order smallint [not null]
  locked boolean [not null, default: false, note: 'Indicates if the category is used in an exam and is locked for editing']
  deleted_at timestamp
}

Table languages {
  id smallint [primary key]
  slug varchar(255) [not null, unique]
  name varchar(255) [not null]
  extension varchar(16) [not null]
  version varchar(16) [not null]
  is_enabled boolean [not null, default: true]
}

Table test_cases {
  id bigint [primary key]
  question_id bigint [not null]
  order smallint [not null]
  content jsonb [not null]
  data_hash varchar(64) [not null]
  hidden boolean [not null, default: false]
  deleted_at timestamp

  // Composite unique key
  indexes {
    (question_id, order) [unique]
  }
}

Table boilerplates {
  id bigint [primary key]
  question_id bigint [not null]
  language_id smallint [not null]
  code text [not null]

  // Composite unique key
  indexes {
    (question_id, language_id) [unique]
  }
}

Table exam_permissions {
  exam_id bigint [primary key]
  user_id bigint [primary key]
  permissions smallint [not null, note: 'Bit flags for READ, WRITE, PARTICIPATE and EVALUATE permissions']
}
