Table users {
  id bigint [primary key]
  username varchar(255) [not null, unique]
  email varchar(255) [not null, unique]
  password varchar(255) [null]
  first_name varchar(255) [null]
  last_name varchar(255) [null]
  verified boolean [not null, default: false]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table exams {
  id bigint [primary key]
  title varchar
  starts_at timestamp [not null]
  ends_at timestamp [not null]
  created_by bigint
  type varchar [note: 'Values can be "LINK" or "INVITE"']
  max_candidates_count integer [note: 'Maximum number of candidates that can take the exam']
  paper_id bigint [note: 'Indicates which paper will be used in this exam']
}

Table papers {
  id bigint [primary key]
  title varchar(255) [not null, default: 'Untitled Paper']
  max_score integer [not null, default: 0, note: 'Maximum score that can be awarded in this paper - Sum of individual scores of each question']
  duration_minutes integer [not null]
  question_counts json [not null, default: '{"mcq":0,"short":0,"long":0}', note: 'Count of each question type']
}

Table paper_ownerships {
  id bigint [primary key]
  user_id bigint [not null]
  paper_id bigint [not null]
  path text [null]
  type varchar(10) [not null, note: 'Can be OWNER or SHARED']
}

Table questions {
  id bigint [primary key]
  question json [not null]
  category_id bigint [not null, note: 'Used to group multiple questions']
  order integer [not null]
  type varchar(20) [not null, note: 'MCQ, Short-answer, Long-answer']
  tags json [not null, default: '[]']
  paper_id bigint [null]
  max_score integer [not null]
  correct_answer text [null]
  locked boolean [not null, default: false, note: 'Indicates if the question is used in an exam and is locked for editing']
}

Table exam_participants {
  id bigint [primary key]
  started_at timestamp
  ended_at timestamp
  scheduled_end_time timestamp [note: 'We cannot rely on calculating (starts_at + paper.duration) everytime because the paper duration could update during the exam']
  exam_id bigint
  user_id bigint [note: 'Which users can take this exam']
  score_awarded integer [note: 'Total score obtained in this exam by this candidate - Sum of individual scores of each answer']
}

Table answers {
  id bigint [primary key]
  exam_participant_id bigint [note: 'Which exam and user this answer-set belongs to']
  question_id bigint [note: 'This answer is for which question']
  answer text
  score_awarded integer [note: 'Score awarded on this answer', default: 0]
  comments text [note: 'Any comments the evaluator might want to add, for e.g. why is the answer wrong']
  evaluated boolean [not null, default: false]
}

Table exam_questions {
  id bigint [primary key]
  exam_id bigint [not null]
  question_id bigint [not null]
}

Table exam_categories {
  id bigint [primary key]
  exam_id bigint [not null]
  category_id bigint [not null]
}

Table sessions {
  key uuid [primary key]
  token varchar(255) [not null]
  expires_at timestamp [not null]
  csrf_token varchar(32) [not null]
}

Table question_categories {
  id bigint [primary key]
  paper_id bigint [null]
  name varchar(255) [not null]
  order integer [not null, note: 'Used to maintain the order of categories']
  locked boolean [not null, default: false, note: 'Indicates if the category is used in an exam and is locked for editing']
}

Table otps {
  email varchar(255) [primary key]
  otp varchar(6) [not null]
  otp_expires_at timestamp [not null]
  purpose integer [not null]
}

// Ref: exams.created_by > users.id // One user can create many exams
// Ref: paper_ownerships.user_id > users.id // Which user owns or shares this paper
Ref: paper_ownerships.paper_id > papers.id // Which paper is owned or shared
Ref: questions.paper_id > papers.id // One exam can have many questions
Ref: questions.category_id > question_categories.id // One category can have many questions
Ref: question_categories.paper_id > papers.id // One paper can have many categories

// Ref: exams.paper_id > papers.id // Many exams can be created using a single paper
Ref: exam_participants.exam_id > exams.id // One exam can have many participants
Ref: exam_participants.user_id > users.id // Which users can take this exam
Ref: answers.exam_participant_id > exam_participants.id // One participant can have many answers
// Ref: answers.question_id > questions.id // One question can have many answers
Ref: exam_questions.exam_id > exams.id // One exam can have many questions
Ref: exam_questions.question_id > questions.id // Link questions to exam
Ref: exam_categories.exam_id > exams.id // One exam can have many categories
Ref: exam_categories.category_id > question_categories.id // Link categories to exam
