Table users {
  id bigint [primary key]
  username varchar(255) [not null, unique]
  email varchar(255) [not null, unique]
  password varchar(255) [null]
  first_name varchar(255) [null]
  last_name varchar(255) [null]
  verified boolean [not null, default: false]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table exams {
  id bigint [primary key]
  title varchar
  starts_at timestamp [not null]
  ends_at timestamp [not null]
  created_by bigint [not null]
  type varchar(16) [default: 'LINK', note: 'Values can be "LINK" or "INVITE"']
  max_candidates_count integer [not null, note: 'Maximum number of candidates that can take the exam']
  duration_minutes integer [not null]
  paper_id bigint [not null, note: 'Indicates which paper will be used in this exam']
  participant_counts json [default: '{"unattended":0,"invited":0,"started":0,"ended":0}']
}

Table papers {
  id bigint [primary key]
  title varchar(255) [not null, default: 'Untitled Paper']
  max_score integer [default: 0, note: 'Maximum score that can be awarded in this paper - Sum of individual scores of each question']
  duration_minutes integer [not null]
  question_counts json [default: '{"mcq":0,"short":0,"long":0}', note: 'Count of each question type']
  created_by bigint [not null]
}

Table permissions {
  paper_id bigint [primary key]
  user_id bigint [primary key]
  permissions smallint [not null, note: 'Bit flags for READ and WRITE permissions']
}

Table questions {
  id bigint [primary key]
  category_id bigint [not null]
  question json [not null]
  order integer [not null]
  type varchar(20) [not null, note: 'MCQ, Short-answer, Long-answer']
  tags json [default: '[]']
  paper_id bigint
  max_score integer [not null]
  correct_answer text
  locked boolean [not null, default: false, note: 'Indicates if the question is used in an exam and is locked for editing']
}

Table exam_participants {
  id bigint [primary key]
  exam_id bigint [not null]
  user_id bigint [not null]
  score_awarded integer [not null, note: 'Total score obtained in this exam by this candidate - Sum of individual scores of each answer']
  status integer [not null, default: 1]
  started_at timestamp
  ended_at timestamp
  scheduled_end_time timestamp [note: 'We cannot rely on calculating (starts_at + paper.duration) everytime because the paper duration could update during the exam']
}

Table answers {
  id bigint [primary key]
  exam_participant_id bigint [not null]
  question_id bigint [not null]
  answer text
  score_awarded integer [not null, default: 0]
  comments text [note: 'Any comments the evaluator might want to add, for e.g. why is the answer wrong']
  evaluated boolean [not null, default: false]
}

Table exam_questions {
  id bigint [primary key]
  exam_id bigint [not null]
  question_id bigint [not null]
  category_id bigint [not null]
  order integer [not null]
}

Table exam_categories {
  id bigint [primary key]
  exam_id bigint [not null]
  category_id bigint [not null]
}

Table sessions {
  key uuid [primary key]
  token varchar(255) [not null]
  expires_at timestamp [not null]
  csrf_token varchar(32) [not null]
}

Table question_categories {
  id bigint [primary key]
  paper_id bigint
  name varchar(255) [not null]
  order integer [not null]
  locked boolean [not null, default: false, note: 'Indicates if the category is used in an exam and is locked for editing']
}

Table otps {
  email varchar(255) [primary key]
  otp varchar(6) [not null]
  otp_expires_at timestamp [not null]
  purpose integer [not null]
}
